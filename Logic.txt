const fetchProducts = async (sessionAttributes) => {
    const { product, upsell } = await getProductsFromTable(sessionAttributes.need, sessionAttributes.o2Customer, (0, utilityChecks_1.isPresent)(sessionAttributes.tvChannels)
        ? sessionAttributes.tvChannels
        : fetchProductsLexModel_1.NOT_APPLICABLE, sessionAttributes.usage);
    const productBundles = await (0, bundlesApiCall_1.fetchProductBundles)({ productId: product.productId, upsellId: upsell.productId }, sessionAttributes.serviceabilityObject, sessionAttributes.availableServices);
    let productDetails = { ...product };
    let upsellDetails = { ...upsell };
    for (const bundle of productBundles.bundles) {
        if (bundle.bundleId === product.productId) {
            productDetails = { ...product, bundle: bundle };
            if (product.productId === upsell.productId) {
                upsellDetails = { ...upsell, bundle: bundle };
            }
            continue;
        }
        upsellDetails = { ...upsell, bundle: bundle };
    }
    return {
        product: productDetails,
        upsell: upsellDetails
    };
};

const getMessages = async (sessionAttributes, products) => {

    sessionAttributes = productMessage.sessionAttributes;
    const skipProductMessages = sessionAttributes.skipProductMessages === "true";
    if (!skipProductMessages) {
        messages = messages
            .concat(productMessage.messages)
            .concat(getActivationFeeMessage(sessionAttributes, products));
    }
    let fallbackPrompt;
    if (sessionAttributes["VM-CR-IDV-PLTFRM"] === "WHATSAPP") {
        const imageUrl = await (0, productsCarousel_1.getImageUrl)(products.product.imageS3Key);
        if (imageUrl !== fetchProductsLexModel_1.NOT_APPLICABLE) {
            const imageUrlMessage = `image-only-message: ${imageUrl}`;
            messages.push(getPlainTextMessage(imageUrlMessage));
        }
        const productImageMessage = (0, productsCarousel_1.getCarouselText)(products.product.bundle, sessionAttributes.waTextPrompt, sessionAttributes.waTitlePrompt);
        messages.push(getPlainTextMessage(productImageMessage.body));
        const waDtmfMessage = getWaDtmfMessage(sessionAttributes.carouselButtonsPrompt, sessionAttributes.preferencePrompt, products);
        messages.push(getPlainTextMessage(waDtmfMessage));
        fallbackPrompt = waDtmfMessage;
        messages = [
            getPlainTextMessage(JSON.stringify({ "array-of-messages": messages }))
        ];
    }
    else {
        const carouselMessage = JSON.stringify(await (0, productsCarousel_1.getProductsCarousel)(products, sessionAttributes));
        fallbackPrompt = carouselMessage;
        messages = messages.concat([
            {
                contentType: lexModel_1.ContentType.CUSTOM_PAYLOAD,
                content: carouselMessage
            }
        ]);
    }
    return {
        messages,
        sessionAttributes: {
            ...sessionAttributes,
            fallbackPrompt
        }
    };
};
const getInitialMessage = (sessionAttributes) => {
    if (sessionAttributes.need.toLowerCase().includes("TV".toLowerCase())) {
        return sessionAttributes.initialPrompt.replace("{need}", "broadband and TV");
    }
    if (sessionAttributes.need.toLowerCase().includes("Phone".toLowerCase())) {
        return sessionAttributes.initialPrompt.replace("{need}", "broadband and phone");
    }
    return sessionAttributes.initialPrompt.replace("{need}", "broadband");
};
const getProductMessage = (sessionAttributes, products) => {
    let message = fetchProductsLexModel_1.NOT_APPLICABLE;
    // One product recommendation
    if (!(0, exports.isUpsellRequired)(products)) {
        logger_1.default.debug("getProductMessage: One product recommendation");
        message =
            sessionAttributes.o2Customer === "Yes"
                ? sessionAttributes.singleO2ExistingPrompt
                : sessionAttributes.singlePrompt;
        message = message.replace("{productName}", products.product.bundle.bundleName);
        return { messages: [getPlainTextMessage(message)], sessionAttributes };
    }
    // Two product recommendations
    logger_1.default.debug("getProductMessage: Two product recommendations");
    const productPrices = (0, exports.getProductPrices)(products.product.bundle);
    const upsellPrices = (0, exports.getProductPrices)(products.upsell.bundle);
    const productSpeed = getProductSpeedInMbps(products.product.bundle.bundleName);
    const upsellSpeed = getProductSpeedInMbps(products.upsell.bundle.bundleName);
    const priceDifferencePerWeek = getPriceDifferencePerWeek(productPrices.contractCost, upsellPrices.contractCost);
    sessionAttributes = {
        "VM-S-NA-BUNDLESUBTOTAL": upsellPrices.contractCost,
        ...sessionAttributes
    };
    if (productSpeed === fetchProductsLexModel_1.NOT_APPLICABLE || upsellSpeed === fetchProductsLexModel_1.NOT_APPLICABLE) {
        message =
            sessionAttributes.o2Customer === "Yes"
                ? sessionAttributes.doubleO2ExistingNoSpeedsPrompt
                : sessionAttributes.doubleNoSpeedsPrompt;
    }
    else {
        message =
            sessionAttributes.o2Customer === "Yes"
                ? sessionAttributes.doubleO2ExistingPrompt
                : sessionAttributes.doublePrompt;
    }
    message = message
        .replace("{productName}", products.product.bundle.bundleName)
        .replace("{extraPerWeek}", priceDifferencePerWeek)
        .replace("{productSpeed}", productSpeed)
        .replace("{upsellSpeed}", upsellSpeed)
        .replace("{upsellName}", products.upsell.bundle.bundleName)
        .replace("{simData}", getSimData(products));
    if (sessionAttributes.o2Customer === "Yes") {
        return { messages: [getPlainTextMessage(message)], sessionAttributes };
    }
    const upsellMessage = sessionAttributes.doubleUpsellPrompt.replace("{upsellTotalCost}", upsellPrices.contractCost);
    return {
        messages: [
            getPlainTextMessage(message),
            getPlainTextMessage(upsellMessage)
        ],
        sessionAttributes
    };
};

const getProductPrices = (productDetails) => {
    const bundleSubtotal = productDetails.totalMonthlyCosts.bundleSubtotal ?? [];
    const priceRiseSteps = productDetails.totalMonthlyCosts.totalPoundsAndPencePriceSteps ?? [];
    const formatDate = (date) => {
        if (!date)
            return "";
        const d = new Date(date);
        return isNaN(d.getTime())
            ? ""
            : d.toLocaleString("en-GB", { month: "long", year: "numeric" });
    };
    // Add all the normal contract price to an array
    const priceSteps = [
        ...bundleSubtotal.map((step) => ({
            contractCost: step.amount,
            contractDuration: step.duration?.toString(),
            contractStartDate: formatDate(step.startDate ?? "")
        })),
        // Add all the price rise info to an array
        ...priceRiseSteps.map((step) => ({
            priceRiseCost: step.amount,
            priceRiseDuration: step.duration?.toString(),
            priceRiseStartDate: formatDate(step.startDate ?? "")
        }))
    ];
    // Assign the values of contract cost, contract duration and fixed cost
    const contractCost = bundleSubtotal[0].amount;
    const contractDuration = bundleSubtotal[0].duration?.toString() ??
        productDetails.contract.term.toString();
    const fixedCost = bundleSubtotal[bundleSubtotal.length - 1].amount;
    const hasOffer = bundleSubtotal.length > 2;
    return {
        contractCost,
        contractDuration,
        fixedCost,
        hasOffer,
        priceSteps
    };
};
exports.getProductPrices = getProductPrices;
const getPriceDifferencePerWeek = (productPrice, upsellPrice) => {
    const productPriceFloat = parseFloat(productPrice);
    const upsellPriceFloat = parseFloat(upsellPrice);
    if (productPriceFloat && upsellPriceFloat) {
        return ((upsellPriceFloat - productPriceFloat) / 4).toFixed(2);
    }
    return "0";
};
const getSimData = (products) => {
    let simData = "0";
    const bundleProducts = products.upsell.bundle.products;
    for (const product of bundleProducts) {
        if (product.productCategory === "SIM") {
            const productName = product.productName;
            simData = productName.match(/[1-9]\d?\d?GB/g)?.pop() ?? "0";
        }
    }
    return simData;
};
const getActivationFeeMessage = (sessionAttributes, products) => {
    let activationFee = "0";
    const bundleProducts = products.upsell.bundle.products;
    for (const product of bundleProducts) {
        if (product.productName === fetchProductsLexModel_1.ACTIVATION_FEE_PRODUCT_NAME &&
            (0, utilityChecks_1.isPresent)(product.prices)) {
            let activationFeeUpdateCounter = 0;
            for (const activationFeeProduct of product.prices) {
                if (activationFeeUpdateCounter === 0 ||
                    parseFloat(activationFeeProduct.price.amount) <
                        parseFloat(activationFee)) {
                    activationFee = activationFeeProduct.price.amount;
                    activationFeeUpdateCounter++;
                }
            }
        }
    }
    if (parseFloat(activationFee) === 0) {
        return [getPlainTextMessage(sessionAttributes.activationFeeFreePrompt)];
    }
    return [
        getPlainTextMessage(sessionAttributes.activationFeePrompt.replace("{activationFee}", activationFee))
    ];
};
const getProductSpeedInMbps = (productName) => {
    const regexp = /G[1-9]|Gig[1-9]|M[1-9]\d{1,2}/;
    if (regexp.test(productName)) {
        const speed = regexp.exec(productName)[0];
        if (speed.startsWith("G")) {
            const speedInMbps = parseInt(speed.substring(1, speed.length)) * 1024;
            return `${speedInMbps}`;
        }
        return speed.substring(1, speed.length);
    }
    return fetchProductsLexModel_1.NOT_APPLICABLE;
};
